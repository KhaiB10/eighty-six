import math
import torch
import torch.nn as nn
import torch.nn.functional as F
from models.common import Transformer, Block, LayerNorm, precompute_rope_cos_sin
from models.vqvae import VQVAE
from utils.misc import count_params

class _TiedCodebookHead(nn.Module):
    def __init__(self, n_embd, codebook, learn_scale=True, trainable=False, use_bias=False):
        super().__init__()
        V, d = codebook.shape
        self.proj = nn.Linear(n_embd, d, bias=False)
        self.bias = nn.Parameter(torch.zeros(V)) if use_bias else None
        self.trainable = bool(trainable)
        if self.trainable:
            self.E = nn.Parameter(codebook.detach().clone())
        else:
            self.register_buffer("E", codebook.detach().clone(), persistent=False)
        self.out_features = V
        # Distance form, the inverse of z_to_idx's argmin ||z - E_i||^2: ||z||^2 cancels in the softmax, leaving 2 z.E_i - ||E_i||^2.
        E_sq = (codebook.detach() ** 2).sum(-1)
        if not self.trainable:
            self.register_buffer("E_sq", E_sq, persistent=False)
        self.register_buffer("d_scale", E_sq.mean().clamp(min=1e-12), persistent=False)
        self.log_inv_T = nn.Parameter(torch.tensor(0.0)) if learn_scale \
                         else torch.tensor(0.0)

    def forward(self, x):
        z = self.proj(x)
        E_sq = (self.E ** 2).sum(-1) if self.trainable else self.E_sq
        neg_d2 = (2.0 * (z @ self.E.t()) - E_sq) / self.d_scale
        out = neg_d2 * torch.exp(self.log_inv_T)
        return out if self.bias is None else out + self.bias


class _SharedHeadView(nn.Module):
    def __init__(self, head, bias):
        super().__init__()
        self.head = head
        self.bias = bias

    def forward(self, x):
        out = F.linear(x, self.head.weight)
        return out if self.bias is None else out + self.bias


class _SharedHeads(nn.Module):
    # one projection shared by every level; with per_level_bias each level owns its bias
    def __init__(self, head, n, per_level_bias=False):
        super().__init__()
        self.head = head
        self.n = int(n)
        self.per_level_bias = bool(per_level_bias)
        self.biases = nn.ParameterList([nn.Parameter(torch.zeros(head.out_features)) for _ in range(self.n)]) if per_level_bias else None
        self._views = [_SharedHeadView(head, self.biases[i] if per_level_bias else None) for i in range(self.n)]

    def __getitem__(self, i):
        return self._views[i] if self.per_level_bias else self.head

    def __len__(self):
        return self.n


class NextConceptTransformer(Transformer):

    def __init__(self,
                 vqvae_local,
                 config
                 ):
        super().__init__()
        n_embd = config.n_embd
        n_layer = config.n_layer
        dropout = config.dropout
        bias = config.bias
        vocab_size = config.vocab_size
        attn_pattern = config.attn_pattern if hasattr(config, 'attn_pattern') else 'block_diagonal'
        self.config = config

        # A tuple keeps the frozen VQ-VAE out of the module tree and the state dict.
        self.vqvae_local = (vqvae_local,)

        self.levels = self.vqvae_local[0].vq.levels
        self.L = sum(self.levels)

        init_std = math.sqrt(1 / n_embd)

        self.query_level0 = bool(getattr(config, 'query_level0', False))
        self.l0_ar = bool(getattr(config, 'l0_autoregressive', False)) and self.query_level0
        self._lvl_first = 0 if self.query_level0 else 1
        T = self.L if self.query_level0 else self.L - self.levels[0]
        P = config.prefix_len

        # Per-level RoPE: prefix at [0..P-1], each target chunk reset to [0..l_k-1].
        self.use_rope = bool(getattr(config, 'use_rope', False))
        self.rope_base = int(getattr(config, 'rope_base', 10000))

        if self.use_rope:
            head_dim = n_embd // config.n_head
            levels_target = self.levels[self._lvl_first:]

            pos_full = torch.cat([
                torch.arange(P, dtype=torch.long),
                *[torch.arange(l, dtype=torch.long) for l in levels_target],
            ])
            cos_full, sin_full = precompute_rope_cos_sin(pos_full, head_dim, self.rope_base)
            self.register_buffer('rope_cos', cos_full, persistent=False)
            self.register_buffer('rope_sin', sin_full, persistent=False)

            for k, l_k in enumerate(levels_target):
                pos_k = torch.cat([
                    torch.arange(P, dtype=torch.long),
                    torch.arange(l_k, dtype=torch.long),
                ])
                cos_k, sin_k = precompute_rope_cos_sin(pos_k, head_dim, self.rope_base)
                self.register_buffer(f'rope_cos_gen_{k}', cos_k, persistent=False)
                self.register_buffer(f'rope_sin_gen_{k}', sin_k, persistent=False)
        else:
            T_max = max(P, max(self.levels[self._lvl_first:]))
            self.wpe = nn.Embedding(T_max, n_embd)
            wp_prefix = torch.arange(P, dtype=torch.long).view(1, -1)
            wp_target = torch.cat(
                [torch.arange(l, dtype=torch.long) for l in self.levels[self._lvl_first:]]
            ).view(1, -1)
            self.register_buffer('wp_prefix', wp_prefix)
            self.register_buffer('wp_1L', wp_target)
            nn.init.trunc_normal_(self.wpe.weight, mean=0, std=init_std)

        self.transformer = nn.ModuleDict(dict())

        self.transformer.update(dict(
            wte = nn.Linear(self.vqvae_local[0].embed_dim, n_embd, bias=bias)
        ))

        ctx_cfg = getattr(config, 'ctx_encoder', None)
        self.use_ctx_encoder = bool(getattr(ctx_cfg, 'enabled', False)) if ctx_cfg is not None else False
        self.ctx_pretrained = str(getattr(ctx_cfg, 'pretrained', '') or '') if ctx_cfg is not None else ''
        if self.use_ctx_encoder and self.ctx_pretrained:
            from transformers import T5EncoderModel
            self.ctx_t5 = T5EncoderModel.from_pretrained(self.ctx_pretrained)
            self.ctx_t5_finetune = bool(getattr(ctx_cfg, 'finetune', False))
            if not self.ctx_t5_finetune:
                self.ctx_t5.eval()
                for p in self.ctx_t5.parameters():
                    p.requires_grad = False
            d_ctx = int(self.ctx_t5.config.d_model)
            self.ctx_proj = nn.Linear(d_ctx, n_embd, bias=bias)
            self.ctx_pad_id = int(getattr(ctx_cfg, 'pad_id', 0))
            print(f"[ctx_encoder] pretrained={self.ctx_pretrained} d_model={d_ctx}->{n_embd} "
                  f"frozen={not self.ctx_t5_finetune} "
                  f"params={sum(p.numel() for p in self.ctx_t5.parameters())/1e6:.2f}M",
                  flush=True)
        elif self.use_ctx_encoder:

                else:

# Phi (quant_resi): per-level residual conv after upsample, blending (1-r)*h + r*Conv1d(h).
class Phi(nn.Conv1d):
    def __init__(self, embed_dim, quant_resi, learnable_ratio=False):
        ks = 3
        super().__init__(
            in_channels=embed_dim,
            out_channels=embed_dim,
            kernel_size=ks,
            stride=1,
            padding=ks // 2,
        )
        init_ratio = abs(quant_resi)
        self._learnable_ratio = bool(learnable_ratio)
        if self._learnable_ratio:
            r_clamped = min(max(init_ratio, 1e-4), 1 - 1e-4)
            init_logit = math.log(r_clamped / (1 - r_clamped))
            self.ratio_logit = nn.Parameter(torch.tensor(init_logit, dtype=torch.float32))
        else:
            self._ratio_const = init_ratio

    @property
    def resi_ratio(self):
        if self._learnable_ratio:
            return torch.sigmoid(self.ratio_logit)
        return self._ratio_const

    def forward(self, h_BCL):
        r = self.resi_ratio
        return h_BCL * (1 - r) + super().forward(h_BCL) * r


class PhiConvStack(nn.Module):
    def __init__(self, embed_dim, kernel_size=3, hidden_mult=1):
        super().__init__()
        ks = kernel_size
        pad = ks // 2
        h = embed_dim * hidden_mult
        self.c1 = nn.Conv1d(embed_dim, h, kernel_size=ks, padding=pad)
        self.act1 = nn.GELU()
        self.c2 = nn.Conv1d(h, h, kernel_size=ks, padding=pad)
        self.act2 = nn.GELU()
        self.c3 = nn.Conv1d(h, embed_dim, kernel_size=ks, padding=pad)
        nn.init.zeros_(self.c3.weight)
        nn.init.zeros_(self.c3.bias)

    def forward(self, h_BCL):
        y = self.act1(self.c1(h_BCL))
        y = self.act2(self.c2(y))
        y = self.c3(y)
        return h_BCL + y


class PhiShared(nn.Module):
    def __init__(self, qresi):
        super().__init__()
        self.qresi = qresi

    def __getitem__(self, _):
        return self.qresi


class PhiPartiallyShared(nn.Module):
    def __init__(self, qresi_ls):
        super().__init__()
        self.qresi_ls = qresi_ls
        K = len(qresi_ls)
        self.ticks = np.linspace(1 / 3 / K, 1 - 1 / 3 / K, K) if K == 4 else np.linspace(1 / 2 / K, 1 - 1 / 2 / K, K)

    def __getitem__(self, at_from_0_to_1):
        return self.qresi_ls[np.argmin(np.abs(self.ticks - at_from_0_to_1)).item()]

    def extra_repr(self):
        return f'ticks={self.ticks}'


class PhiNonShared(nn.ModuleList):
    def __init__(self, qresi):

        K = len(qresi)
        self.ticks = np.linspace(1 / 3 / K, 1 - 1 / 3 / K, K) if K == 4 else np.linspace(1 / 2 / K, 1 - 1 / 2 / K, K)

    def __getitem__(self, at_from_0_to_1):
        return super().__getitem__(np.argmin(np.abs(self.ticks - at_from_0_to_1)).item())

        return f'ticks={self.ticks}'


def _cfg_get(cfg, key, default):
    if cfg is None:
        return default
    if hasattr(cfg, key):
        return getattr(cfg, key, default)
    if isinstance(cfg, dict) and key in cfg:
        return cfg[key]
    return default


def _build_quant_resi(embed_dim, num_levels, quant_resi_config):
    enabled = bool(_cfg_get(quant_resi_config, 'enabled', True))
    if not enabled:
        return None
    raw_ratio = _cfg_get(quant_resi_config, 'ratio', 0.5)
    share_mode = int(_cfg_get(quant_resi_config, 'share_mode', 0))
    learnable_ratio = bool(_cfg_get(quant_resi_config, 'learnable_ratio', False))
    arch = str(_cfg_get(quant_resi_config, 'arch', 'blend'))
    hidden_mult = int(_cfg_get(quant_resi_config, 'hidden_mult', 1))
    kernel_size = int(_cfg_get(quant_resi_config, 'kernel_size', 3))

    if isinstance(raw_ratio, (list, tuple)) or (hasattr(raw_ratio, '__len__') and not isinstance(raw_ratio, str)):
        ratios_per_level = [float(r) for r in raw_ratio]
        if share_mode != 0:
            raise ValueError(
                f"per-level ratio list requires share_mode=0 (non-shared); got share_mode={share_mode}"
            )
        if len(ratios_per_level) != num_levels:
            raise ValueError(
                f"ratio list must have length {num_levels} (one per level); got {len(ratios_per_level)}"
            )
    else:
        ratios_per_level = [float(raw_ratio)] * num_levels

    def make_phi(r):
        if arch == 'convstack':
            return PhiConvStack(embed_dim, kernel_size=kernel_size, hidden_mult=hidden_mult)
        if learnable_ratio:
            return Phi(embed_dim, r, learnable_ratio=True)
        return Phi(embed_dim, r) if abs(r) > 1e-6 else nn.Identity()

    if share_mode == 0:
        return PhiNonShared([make_phi(r) for r in ratios_per_level])
    if share_mode == 1:
        return PhiShared(make_phi(ratios_per_level[0]))
    return PhiPartiallyShared(nn.ModuleList([make_phi(ratios_per_level[0]) for _ in range(share_mode)]))

def _make_learned_down_l1(D, stride=4):
    conv = nn.Conv1d(D, D, kernel_size=stride, stride=stride, bias=True)
    with torch.no_grad():
        conv.weight.zero_()
        eye_idx = torch.arange(D)
        conv.weight[eye_idx, eye_idx, :] = 1.0 / stride
        conv.bias.zero_()
    return conv


def _make_learned_up_l1(D, stride=4):
    convT = nn.ConvTranspose1d(D, D, kernel_size=stride, stride=stride, bias=True)
    with torch.no_grad():
        convT.weight.zero_()
        eye_idx = torch.arange(D)
        convT.weight[eye_idx, eye_idx, :] = 1.0
        convT.bias.zero_()
    return convT


def _plan_cascade_strides(total_stride, base=4):
    S = int(total_stride)
    strides = []
    while S > 1:
        f = min(base, S)
        while f > 1 and S % f != 0:
            f -= 1
        if f <= 1:
            strides.append(S)
            S = 1
        else:
            strides.append(f)
            S //= f
    return strides or [1]


def _cascade_norm(D, groups=8):
    return nn.GroupNorm(groups, D, affine=False)


def _make_cascaded_down_l1(D, total_stride, base=4, inter_norm=False, final_norm=False):
    strides = _plan_cascade_strides(total_stride, base)
    mods = []
    for i, s in enumerate(strides):
        if inter_norm and i > 0:
            mods.append(_cascade_norm(D))
        mods.append(_make_learned_down_l1(D, stride=s))
    if final_norm:
        mods.append(_cascade_norm(D))
    return nn.Sequential(*mods)


def _make_cascaded_up_l1(D, total_stride, base=4, inter_norm=False):
    strides = list(reversed(_plan_cascade_strides(total_stride, base)))
    mods = []
    for i, s in enumerate(strides):
        if inter_norm and i > 0:
            mods.append(_cascade_norm(D))
        mods.append(_make_learned_up_l1(D, stride=s))
    return nn.Sequential(*mods)


def _make_zeroinit_residual_conv(D, kernel_size=3):
    pad = kernel_size // 2
    conv = nn.Conv1d(D, D, kernel_size=kernel_size, stride=1, padding=pad, bias=True)
    with torch.no_grad():
        conv.weight.zero_()
        conv.bias.zero_()
    return conv

import torch
import torch.nn as nn
from torch.nn import functional as F
import math

class LayerNorm(nn.Module):
    def __init__(self, ndim, bias=False):
        super().__init__()
        self.weight = nn.Parameter(torch.ones(ndim))
        self.bias = nn.Parameter(torch.zeros(ndim)) if bias else None

    def forward(self, input):
        return F.layer_norm(input, self.weight.shape, self.weight, self.bias, 1e-6)


class RMSNorm(nn.Module):
    def __init__(self, dim, eps=1e-6):
        super().__init__()
        self.weight = nn.Parameter(torch.ones(dim))
        self.eps = eps

    def forward(self, x):
        orig_dtype = x.dtype
        x_f = x.float()
        rms = x_f.pow(2).mean(dim=-1, keepdim=True).add_(self.eps).rsqrt_()
        return (x_f * rms).to(orig_dtype) * self.weight


class SelfAttention(nn.Module):
    def __init__(self, config):
        super().__init__()
        assert config.n_embd % config.n_head == 0
        self.c_attn = nn.Linear(config.n_embd, 3 * config.n_embd, bias=config.bias)
        self.c_proj = nn.Linear(config.n_embd, config.n_embd, bias=config.bias)
        self.attn_dropout = nn.Dropout(config.dropout)
        self.resid_dropout = nn.Dropout(config.dropout)
        self.n_head = config.n_head
        self.n_embd = config.n_embd
        self.dropout = config.dropout
        self.use_qk_norm = bool(getattr(config, "use_qk_norm", False))
        if self.use_qk_norm:
            head_dim = config.n_embd // config.n_head
            self.q_norm = RMSNorm(head_dim)
            self.k_norm = RMSNorm(head_dim)
        # attn_window=w restricts attention to |i-j| <= w; 0 means full.
        self.attn_window = int(getattr(config, "attn_window", 0) or 0)
        assert hasattr(torch.nn.functional, "scaled_dot_product_attention")

    def _window_mask(self, T, device):
        i = torch.arange(T, device=device)
        return (i[:, None] - i[None, :]).abs() <= self.attn_window

    def forward(self, x):
        B, T, C = x.size()
        q, k, v = self.c_attn(x).split(self.n_embd, dim=2)
        k = k.view(B, T, self.n_head, C // self.n_head).transpose(1, 2)
        q = q.view(B, T, self.n_head, C // self.n_head).transpose(1, 2)
        v = v.view(B, T, self.n_head, C // self.n_head).transpose(1, 2)

        if self.use_qk_norm:
            q = self.q_norm(q)
            k = self.k_norm(k)

        attn_mask = self._window_mask(T, x.device) if self.attn_window > 0 else None
        y = torch.nn.functional.scaled_dot_product_attention(
            q, k, v, attn_mask=attn_mask, dropout_p=self.dropout if self.training else 0
        )
        y = y.transpose(1, 2).contiguous().view(B, T, C)
        y = self.resid_dropout(self.c_proj(y))
        return y


class MLP(nn.Module):
    def __init__(self, config):
        super().__init__()
        self.c_fc = nn.Linear(config.n_embd, 4 * config.n_embd, bias=config.bias)
        self.gelu = nn.GELU()
        self.c_proj = nn.Linear(4 * config.n_embd, config.n_embd, bias=config.bias)
        self.dropout = nn.Dropout(config.dropout)

    def forward(self, x):
        x = self.c_fc(x)
        x = self.gelu(x)
        x = self.c_proj(x)
        x = self.dropout(x)
        return x


class Block(nn.Module):
    def __init__(self, config):
        super().__init__()
        self.ln_1 = LayerNorm(config.n_embd, bias=config.bias)
        self.attn = SelfAttention(config)
        self.ln_2 = LayerNorm(config.n_embd, bias=config.bias)
        self.mlp = MLP(config)

    def forward(self, x):
        x = x + self.attn(self.ln_1(x))
        x = x + self.mlp(self.ln_2(x))
        return x

class Encoder(nn.Module):
    def __init__(
        self,
        in_vocab_size,
        context_length,
        embed_dim,
        n_layers,
        compression_factor=4,
        kernel_size=8,
        stride=2,
        bias=False,
        padding=3,
        dropout=0.1,
        enc_layers=0,
        use_qk_norm=False,
        enc_attn_window=0,
        enc_conv_layers=0,
        enc_conv_kernel=3,
        use_pos_emb=True,
    ):
        super().__init__()

        assert compression_factor ==1 or compression_factor % 2 == 0, "Compression factor must be a power of 2"
        self.compression_factor = compression_factor
        self.wte = nn.Embedding(in_vocab_size, embed_dim)
        self.wpe = nn.Embedding(context_length, embed_dim) if use_pos_emb else None
        self.drop = nn.Dropout(dropout)

        self.blocks = nn.ModuleList([
            Block(config=type("Config", (object,), {
                "n_embd": embed_dim, "n_head": n_layers, "bias": True,
                "dropout": dropout, "use_qk_norm": use_qk_norm,
                "attn_window": int(enc_attn_window)}))
            for _ in range(int(enc_layers))
        ])
        k = int(enc_conv_kernel)
        self.conv_blocks = nn.ModuleList()
        for _ in range(int(enc_conv_layers)):
            c1 = nn.Conv1d(embed_dim, embed_dim, k, padding=k // 2)
            c2 = nn.Conv1d(embed_dim, embed_dim, k, padding=k // 2)
            nn.init.zeros_(c2.weight); nn.init.zeros_(c2.bias)
            self.conv_blocks.append(nn.ModuleDict({
                "ln": LayerNorm(embed_dim), "c1": c1, "c2": c2}))
        self.ln_e = LayerNorm(embed_dim) if (int(enc_layers) > 0 or int(enc_conv_layers) > 0) else None

        # Sequence compression: f>1 shortens the sequence before quantization; init is an average pool.
        self.down = None
        if int(compression_factor) > 1:
            f = int(compression_factor)
            conv = nn.Conv1d(embed_dim, embed_dim, kernel_size=f, stride=f, bias=True)
            with torch.no_grad():
                conv.weight.zero_()
                idx = torch.arange(embed_dim)
                conv.weight[idx, idx, :] = 1.0 / f
                conv.bias.zero_()
            self.down = conv

    def _conv_stack(self, x):
        for blk in self.conv_blocks:
            h = blk["ln"](x).transpose(1, 2)
            h = blk["c2"](F.gelu(blk["c1"](h))).transpose(1, 2)
            x = x + h
        return x

    def forward(self, idx):
        tok_emb = self.wte(idx)
        if self.wpe is not None:
            pos = torch.arange(0, idx.size(1), dtype=torch.long, device=idx.device)
            tok_emb = tok_emb + self.wpe(pos)
        x = self.drop(tok_emb)
        x = self._conv_stack(x)
        for blk in self.blocks:
            x = blk(x)
        if self.ln_e is not None:
            x = self.ln_e(x)
        if self.down is not None:
            x = self.down(x.transpose(1, 2)).transpose(1, 2)
        return x

    def forward_nopos(self, idx):
        B, T, C = idx.shape[0], idx.shape[1], self.wte.embedding_dim
        pos = torch.arange(T, device=idx.device, dtype=torch.float32).unsqueeze(1)
        i = torch.arange(0, C, 2, device=idx.device, dtype=torch.float32)
        div = torch.exp(-math.log(10000.0) * i / C)
        pe = torch.zeros(T, C, device=idx.device, dtype=torch.float32)
        pe[:, 0::2] = torch.sin(pos * div)
        pe[:, 1::2] = torch.cos(pos * div)
        x = self.drop(self.wte(idx) + pe.to(self.wte.weight.dtype).unsqueeze(0))
        x = self._conv_stack(x)
        for blk in self.blocks:
            x = blk(x)
        if self.ln_e is not None:
            x = self.ln_e(x)
        return x


class Decoder(nn.Module):
    def __init__(
        self,
        in_vocab_size,
        embed_dim,
        n_layers,
        compression_factor=4,
        kernel_size=8,
        stride=2,
        bias=False,
        padding=3,
        dropout=0.1,
        use_qk_norm=False,
        dec_layers=1,
        use_pos_emb=False,
        context_length=None,
    ):
        super().__init__()

        assert compression_factor ==1 or compression_factor % 2 == 0, "Compression factor must be a power of 2"

        self.wpe = nn.Embedding(int(context_length), embed_dim) if use_pos_emb else None
        self.drop = nn.Dropout(dropout)
        self.ln_f = LayerNorm(embed_dim)
        self.compression_factor = compression_factor

        # n_layers is the head count here, not a depth.
        _cfg = type("Config", (object,), {
            "n_embd": embed_dim, "n_head": n_layers, "bias": True,
            "dropout": dropout, "use_qk_norm": use_qk_norm})
        self.middle_block = Block(config=_cfg)
        self.extra_blocks = nn.ModuleList(
            [Block(config=_cfg) for _ in range(max(0, int(dec_layers) - 1))]
        )

        self.output_layer = nn.Linear(embed_dim, in_vocab_size, bias=False)
        self.ln_f = LayerNorm(embed_dim)

        self.up = None
        if int(compression_factor) > 1:
            f = int(compression_factor)
            convT = nn.ConvTranspose1d(embed_dim, embed_dim, kernel_size=f, stride=f, bias=True)
            with torch.no_grad():
                convT.weight.zero_()
                idx = torch.arange(embed_dim)
                convT.weight[idx, idx, :] = 1.0
                convT.bias.zero_()
            self.up = convT


    def forward(self, z):
        if self.up is not None:
            z = self.up(z.transpose(1, 2)).transpose(1, 2)
        if self.wpe is not None:
            z = z + self.wpe(torch.arange(z.shape[1], dtype=torch.long, device=z.device))
        x = self.middle_block(z)
        for blk in self.extra_blocks:
            x = blk(x)
        x = self.ln_f(x)
        x = self.output_layer(x)
        return x
