    def _encode_context(self, prefix_ids):
        if self.ctx_pretrained:
            am = (prefix_ids != self.ctx_pad_id).long()
            if self.ctx_t5_finetune:
                h = self.ctx_t5(input_ids=prefix_ids, attention_mask=am).last_hidden_state
            else:
                with torch.no_grad():
                    h = self.ctx_t5(input_ids=prefix_ids,
                                    attention_mask=am).last_hidden_state
            return self.ctx_proj(h.to(self.ctx_proj.weight.dtype))
        h = self.ctx_wte(prefix_ids)
        rope_p = (self.rope_cos[:self.prefix_len],
                  self.rope_sin[:self.prefix_len]) if self.use_rope else None
        P = self.prefix_len
        ctx_mask = torch.zeros(1, 1, P, P, device=h.device)
        for block in self.ctx_blocks:
            h = block(h, attn_mask=ctx_mask, rope_cos_sin=rope_p)
        return self.ctx_ln(h)

    def forward(self, x, prefix_ids=None, prefix_pad_mask=None):
        if self.use_ctx_encoder:
            assert prefix_ids is not None, "ctx_encoder requires prefix_ids"
            prefix_emb = self._encode_context(prefix_ids)
            x_target = x
            if self.use_post_upsample_conv:
                x_target = self._apply_post_upsample_convs_multi(x_target)
            x = self.transformer.wte(x_target)
            if not self.use_rope:
                x = x + self.wpe(self.wp_1L)
            if self.config.use_level_encoding:
                x = x + self.transformer.wle(self.lvl_1L)
            return self._run_stack(prefix_emb, x, prefix_pad_mask)

        z_prefix = x[:, :self.prefix_len, :]
        x_target = x[:, self.prefix_len:, :]

        if self.use_post_upsample_conv:
            x_target = self._apply_post_upsample_convs_multi(x_target)

        prefix_emb = self.transformer.wte(z_prefix)
        x = self.transformer.wte(x_target)

        if not self.use_rope:
            prefix_emb = prefix_emb + self.wpe(self.wp_prefix)
            x = x + self.wpe(self.wp_1L)

        if self.config.use_level_encoding:
            x += self.transformer.wle(self.lvl_1L)

        return self._run_stack(prefix_emb, x, prefix_pad_mask)

    def idx_to_zhats(self, idx, return_zlevels=True):
        z_hats = []
        z_levels = []
        z_hat = torch.zeros(
            idx.shape[0], self.levels[-1], self.embed_dim, device=idx.device
        )

        level_accum = 0
        for level_idx in range(len(self.levels)):
            emb_i = self.embeddings[level_idx]
            emb_at_level = emb_i(idx[:, level_accum : level_accum + self.levels[level_idx]])
            if level_idx != len(self.levels) - 1:
                z_q_level = self._up_to_full(emb_at_level, level_idx, self.levels[-1])
            else:
                z_q_level = emb_at_level
            z_q_level = self._apply_quant_resi(z_q_level, level_idx)
            z_hat.add_(z_q_level)
            z_hats.append(z_hat)
            if return_zlevels:
                z_levels.append(
                    F.interpolate(
                        z_hat.transpose(-1, -2),
                        size=(self.levels[level_idx],),
                        mode="area",
                    ).transpose(-1, -2)
                    if level_idx != len(self.levels) - 1
                    else z_hat
                )
            level_accum += self.levels[level_idx]

        results = (z_hats,)
        if return_zlevels:
            results = results + (z_levels,)
        return results

def pick_tokens(logits):
    if freq <= 0 and not ADJ:
        return logits.argmax(-1)
    B, T, V = logits.shape
    lg = logits.double()
    allow = (ALLOW[:V] * T)[None, :] if ALLOW is not None else None
    al = ADJ[0] if ADJ else 0.0
    aw = int(ADJ[1]) if ADJ and len(ADJ) > 1 else 1
    ad = ADJ[2] if ADJ and len(ADJ) > 2 else 1.0
    cnt = torch.zeros(B, V, dtype=torch.float64, device=lg.device)
    out = torch.empty(B, T, dtype=torch.long, device=lg.device)
    hist = []
    for t in range(T):
        step = lg[:, t, :].clone()
        if freq > 0:
            excess = (cnt - allow).clamp_min(0.0) if allow is not None else cnt
            step -= freq * excess
            if mu > 0 and allow is not None:
                step += mu * ((T - t) / T) * (allow - cnt).clamp_min(0.0)
        for j, h in enumerate(reversed(hist[-aw:])):
            step.scatter_add_(1, h[:, None], torch.full((B, 1), -al * ad ** j, dtype=torch.float64, device=lg.device))
        p = step.argmax(-1)
        out[:, t] = p
        hist.append(p)
        cnt.scatter_add_(1, p[:, None], torch.ones(B, 1, dtype=torch.float64, device=lg.device))
    return out
